#include <stdio.h>
#include <string.h>

struct student
{
    char name[80];
    int rollno, age;
    int marks[5];
    char section;
};

void printDetails(struct student s[], int n);
void calculateAvg(struct student s[], int n);
void calculateGrade(struct student s[], int n);
void calculateCGPA(struct student s[], int n);

int main()
{
    int i, j, n, choice;
    struct student s[80];

    printf("Enter number of students: ");
    scanf("%d", &n);
    getchar();   // clear newline

    for (i = 0; i < n; i++)
    {
        printf("\nEnter details of student %d\n", i + 1);

        printf("Enter name: ");
        fgets(s[i].name, sizeof(s[i].name), stdin);
        s[i].name[strcspn(s[i].name, "\n")] = '\0';

        printf("Enter roll number: ");
        scanf("%d", &s[i].rollno);

        printf("Enter age: ");
        scanf("%d", &s[i].age);

        printf("Enter section: ");
        getchar();
        scanf("%c", &s[i].section);

        printf("Enter marks (5 subjects): ");
        for (j = 0; j < 5; j++)
        {
            scanf("%d", &s[i].marks[j]);
        }
        getchar();
    }

    do
    {
        printf("\n<---- MENU ---->\n");
        printf("1. Print student details\n");
        printf("2. Calculate average marks\n");
        printf("3. Calculate grade\n");
        printf("4. Calculate CGPA\n");
        printf("5. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
        case 1:
            printDetails(s, n);
            break;
        case 2:
            calculateAvg(s, n);
            break;
        case 3:
            calculateGrade(s, n);
            break;
        case 4:
            calculateCGPA(s, n);
            break;
        case 5:
            printf("Exiting program...\n");
            break;
        default:
            printf("Invalid choice\n");
        }
    } while (choice != 5);

    return 0;
}

void printDetails(struct student s[], int n)
{
    int i, j;
    for (i = 0; i < n; i++)
    {
        printf("\nStudent %d\n", i + 1);
        printf("Name: %s\n", s[i].name);
        printf("Roll No: %d\n", s[i].rollno);
        printf("Age: %d\n", s[i].age);
        printf("Section: %c\n", s[i].section);
        printf("Marks: ");
        for (j = 0; j < 5; j++)
        {
            printf("%d ", s[i].marks[j]);
        }
        printf("\n");
    }
}

void calculateAvg(struct student s[], int n)
{
    int i, j;
    float sum, avg;

    for (i = 0; i < n; i++)
    {
        sum = 0;
        for (j = 0; j < 5; j++)
        {
            sum += s[i].marks[j];
        }
        avg = sum / 5;
        printf("Average marks of %s = %.2f\n", s[i].name, avg);
    }
}

void calculateGrade(struct student s[], int n)
{
    int i, j;
    float sum, avg;
    char grade;

    for (i = 0; i < n; i++)
    {
        sum = 0;
        for (j = 0; j < 5; j++)
        {
            sum += s[i].marks[j];
        }

        avg = sum / 5;

        if (avg >= 90)
            grade = 'A';
        else if (avg >= 75)
            grade = 'B';
        else if (avg >= 60)
            grade = 'C';
        else if (avg >= 40)
            grade = 'D';
        else
            grade = 'F';

        printf("Grade of %s = %c\n", s[i].name, grade);
    }
}

void calculateCGPA(struct student s[], int n)
{
    int i, j;
    float sum, cgpa;

    for (i = 0; i < n; i++)
    {
        sum = 0;
        for (j = 0; j < 5; j++)
        {
            sum += s[i].marks[j];
        }
        cgpa = (sum / 5) / 10;
        printf("CGPA of %s = %.2f\n", s[i].name, cgpa);
    }
}
